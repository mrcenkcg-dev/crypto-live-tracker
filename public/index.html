// Dual-Vacancy Logic & 7-Day Test State const CONFIG = { maxFreeSlots: 1500, microFeeEntry: 0.50, // 50p toll starting at 1501 userSlotLimit: 10, // Strict cap per participant testWindowDays: 7 }; function evaluateRegistration(currentCount) { if (currentCount <= CONFIG.maxFreeSlots) { return { status: "FREE_TIER", fee: 0.00, remainingFree: CONFIG.maxFreeSlots - currentCount }; } else { return { status: "MICRO_FEE_ACTIVE", fee: CONFIG.microFeeEntry, notice: "Community maintenance toll engaged." }; } }
